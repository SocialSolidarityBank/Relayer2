## 4. L5 계약 (UI 레인은 이 형태를 전제로 만든다)

```
sessions.duration_min int null                      -- D4
PATCH /sessions/:id { duration_min? }               -- 기록 저장 payload에 포함
POST  /sessions/:id/revisions { kind: 'memo'|'transcript'|'summary', text }  -- D3, append-only, 감사 'session.revise'
GET   /sessions/:id/revisions → [{id, kind, text, actor, created_at}]
GET   /sessions/:id/detail → { ..., stale: { ai_summary: bool, mismatch: bool } }  -- 원본 수정 뒤 true
PATCH /cases/:id/next-goals { lines: string[] }     -- D5 (서버는 '\n' join, 기존 컬럼)
GET   /consent-copy → 기존 + { editable: bool(admin) }
PUT   /consent-copy/:domain { copy, items[], purpose_text, retention_text, refusal_text }
      → 새 판 `consent-standard-form-v<N+1>`, DB 표 consent_copy(domain, version, ..., created_by)  -- D6
GET   /users/:id/cases → [{case_id, name, program, seq, next_at}]            -- J1
GET   /assign/cases?q&program&page → { items:[{case_id,name,login,program,seq,phone,email,assignees:[{id,name}]}], total }
PUT   /cases/:id/assignments { user_ids: number[] }  -- 전체 치환, 감사 'assignment.set'  -- J3, D7
POST  /assignment-requests/:id/approve → 승인 + case_assignments insert (J4)
```
